function fixingError(value) {
    if (value >= 4.00 && value < 5.00) {
        return value - 0.02;
    }
    if (value >= 5.00 && value < 8.00) {
        return value - 0.03;
    }
    if (value >= 8.00 && value < 10.00) {
        return value - 0.04;
    }
    if (value >= 10.00 && value < 13.00) {
        return value - 0.06;
    }
    if (value >= 13.00 && value < 17.00) {
        return value - 0.05;
    }
    if (value >= 17.00 && value < 19.00) {
        return value - 0.04;
    }
    if (value >= 19.00 && value < 21.00) {
        return value - 0.02;
    }
}


function AmpereVal ( value){
    let val ; 
    value = value-4.00;
    val = 16 / value;
    val = 5/val; 
    val = val * 6; 
    return val;
}

function PressureVal ( value){
    let val; 
    value = value-4.00;
    val = 16 / value;
    val = 60/ val;
    val = val * 14.504;
    return val; 
}

function TemperatureVal (value){
    let val; 
    value = value-4.00;
    val = 16/ value;

}
